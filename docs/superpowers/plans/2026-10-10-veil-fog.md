```java
package com.pocketkingdoms.client.veil;

/**
 * The Veil fog's noise, the same as the study's: 32 x 32 x 32 tileable value noise (3 octaves, period 4 lattice cells),
 * two fields (seeds 71 and 83). The game's renderer has no 3D textures, so the 32 slices (along the depth axis) sit side
 * by side in one 2D atlas, each 34 pixels wide: one copied edge pixel each side, so linear filtering never blends into
 * the next slice. A pixel of slice k holds (a[k], a[k+1], b[k], b[k+1]): one read gives a field at two neighbouring
 * slices, and the shader blends them. Plain logic, unit tested; VeilFogRenderer uploads it.
 */
public final class VeilNoise {

    public static final int SIZE = 32, TILE = SIZE + 2, ATLAS_W = SIZE * TILE, ATLAS_H = SIZE;
    private static final int SEED_A = 71, SEED_B = 83, PERIOD = 4;

    private VeilNoise() {}

    /** A repeatable hash of a lattice point, in [0, 1] (the study's hsh, bit for bit). */
    static double hash(int i, int j, int k, int seed) {
        int n = (i * 374761393) ^ (j * 668265263) ^ (k * 1274126177) ^ (seed * 1442695041);
        n = (n ^ (n >>> 13)) * 1274126177;
        n ^= n >>> 16;
        return (n & 0xFFFFFFFFL) / 4294967295.0;
    }

    private static double smooth(double t) { return t * t * (3 - 2 * t); }

    private static int wrap(int a, int period) { return Math.floorMod(a, period); }

    /** Value noise on a lattice that repeats every `period` cells. */
    private static double periodic(double x, double y, double z, int period, int seed) {
        int xi = (int) Math.floor(x), yi = (int) Math.floor(y), zi = (int) Math.floor(z);
        double fx = smooth(x - xi), fy = smooth(y - yi), fz = smooth(z - zi);
        double[] c = new double[8];
        for (int n = 0; n < 8; n++)
            c[n] = hash(wrap(xi + (n & 1), period), wrap(yi + ((n >> 1) & 1), period), wrap(zi + ((n >> 2) & 1), period), seed);
        double x00 = c[0] + (c[1] - c[0]) * fx, x10 = c[2] + (c[3] - c[2]) * fx;
        double x01 = c[4] + (c[5] - c[4]) * fx, x11 = c[6] + (c[7] - c[6]) * fx;
        double y0 = x00 + (x10 - x00) * fy, y1 = x01 + (x11 - x01) * fy;
        return y0 + (y1 - y0) * fz;
    }

    /** Three octaves, each still repeating every 4 units. */
    static double fbm(double x, double y, double z, int seed) {
        double sum = 0, amp = 0.5, norm = 0;
        for (int o = 0, f = 1; o < 3; o++, f *= 2) {
            sum += amp * periodic(x * f, y * f, z * f, PERIOD * f, seed + o * 17);
            norm += amp;
            amp *= 0.5;
        }
        return sum / norm;
    }

    /** One field, stretched to 0..255; index (z * SIZE + y) * SIZE + x. */
    public static int[] field(int seed) {
        double[] v = new double[SIZE * SIZE * SIZE];
        double lo = Double.MAX_VALUE, hi = -Double.MAX_VALUE;
        for (int z = 0; z < SIZE; z++) for (int y = 0; y < SIZE; y++) for (int x = 0; x < SIZE; x++) {
            double n = fbm(x / 8.0, y / 8.0, z / 8.0, seed);
            v[(z * SIZE + y) * SIZE + x] = n;
            lo = Math.min(lo, n);
            hi = Math.max(hi, n);
        }
        int[] out = new int[v.length];
        for (int i = 0; i < v.length; i++) out[i] = (int) Math.round((v[i] - lo) / (hi - lo) * 255);
        return out;
    }

    /** The atlas as ARGB pixels, ATLAS_W x ATLAS_H, row-major. */
    public static int[] atlas() {
        int[] a = field(SEED_A), b = field(SEED_B), out = new int[ATLAS_W * ATLAS_H];
        for (int k = 0; k < SIZE; k++) {
            int k2 = (k + 1) % SIZE;
            for (int y = 0; y < SIZE; y++) for (int c = 0; c < TILE; c++) {
                int x = Math.floorMod(c - 1, SIZE), i = (k * SIZE + y) * SIZE + x, i2 = (k2 * SIZE + y) * SIZE + x;
                out[y * ATLAS_W + k * TILE + c] = (b[i2] << 24) | (a[i] << 16) | (a[i2] << 8) | b[i];
            }
        }
        return out;
    }
}
```

```java
package com.pocketkingdoms.client.veil;

/**
 * The Veil fog's closed forms, as in veil_fog.fsh (keep the two in step): smoke density falls off as e^(-d/H) with d the
 * distance from a wall's face into the realm, so the smoke along a straight view has a formula and needs no steps. A
 * view is described per wall by dc (the camera's d) and rx (how much d changes per block along the view). Plain logic,
 * unit tested; the renderer uses the top fade, the rest pins down what the shader computes.
 */
public final class VeilFogMath {

    public static final double H = 2.4, SIGMA = 0.8, XMAX = 24, NEAR = 10;

    private VeilFogMath() {}

    /** The smoke between ta and tb along the view: the integral of e^(-d/H). */
    public static double amount(double dc, double rx, double ta, double tb) {
        if (tb <= ta) return 0;
        double da = dc + rx * ta, db = dc + rx * tb;
        if (Math.abs(rx) < 1e-3) return Math.exp(-da / H) * (tb - ta);
        return (-H / rx) * (Math.exp(-db / H) - Math.exp(-da / H));
    }

    /** Where along ta..tb half the smoke lies in front and half behind: where the shader takes colour and fog. */
    public static double middle(double dc, double rx, double ta, double tb) {
        if (Math.abs(rx) < 1e-3 || tb <= ta) return 0.5 * (ta + tb);
        double da = dc + rx * ta, db = dc + rx * tb;
        double dm = -H * Math.log(0.5 * (Math.exp(-da / H) + Math.exp(-db / H)));
        return Math.max(ta, Math.min(tb, (dm - dc) / rx));
    }

    /** The part of ta..tb where lo < d < hi, as {from, to}; empty when to <= from. */
    public static double[] clip(double lo, double hi, double dc, double rx, double ta, double tb) {
        if (Math.abs(rx) < 1e-4) return dc >= lo && dc <= hi ? new double[] {ta, tb} : new double[] {0, -1};
        double a = (lo - dc) / rx, b = (hi - dc) / rx;
        return new double[] {Math.max(ta, Math.min(a, b)), Math.min(tb, Math.max(a, b))};
    }

    /** The smoke starts thinning out here: the study's y 170 in a world from 0 to 320, scaled to this world. */
    public static double fadeStart(int minY, int height) { return minY + height * 170.0 / 320; }

    /** And is gone here (the study's y 290). */
    public static double fadeEnd(int minY, int height) { return minY + height * 290.0 / 320; }
}
```

```java
import org.jspecify.annotations.Nullable;

import java.util.Arrays;
import java.util.Locale;

/**
 * /pd test veilTime: collects the Veil fog pass's GPU time per frame (from timer queries) for a few seconds, then
 * reports once, in milliseconds and as a share of a 60 fps frame, so it means the same on any computer. Plain logic,
 * unit tested; VeilFogRenderer feeds it.
 */
public final class VeilFogTimer {

    private static final double FRAME_60_MS = 1000.0 / 60;

    /** The result: frames measured, GPU time per frame (average, 95th percentile, worst). */
    public record Report(double seconds, int frames, double averageMs, double p95Ms, double worstMs) {
        public String describe() {
            if (frames == 0)
                return String.format(Locale.ROOT, "Veil fog: not drawn in %.1f s (stand in a realm with the Veil in view).", seconds);
            return String.format(Locale.ROOT,
                    "Veil fog over %.1f s, %d frames: %.2f ms of graphics card time a frame on average (95%% under %.2f ms, "
                            + "worst %.2f ms). That is %.1f%% of a 60 fps frame.",
                    seconds, frames, averageMs, p95Ms, worstMs, averageMs / FRAME_60_MS * 100);
        }
    }

    private long[] nanos = new long[256];
    private long until;
    private int frames;
    private double seconds;
    private boolean measuring;

    /** Starts a window of `seconds` from `now` (System.nanoTime), dropping any window still open. */
    public void start(double seconds, long now) {
        this.seconds = seconds;
        until = now + (long) (seconds * 1e9);
        frames = 0;
        measuring = true;
    }

    public boolean measuring() { return measuring; }

    /** One frame's GPU time. Ignored outside a window. */
    public void frame(long gpuNanos) {
        if (!measuring) return;
        if (frames == nanos.length) nanos = Arrays.copyOf(nanos, frames * 2);
        nanos[frames++] = gpuNanos;
    }

    /** The report once the window has closed at `now`, exactly once; null before that or when nothing is measured. */
    public @Nullable Report poll(long now) {
        if (!measuring || now < until) return null;
        measuring = false;
        if (frames == 0) return new Report(seconds, 0, 0, 0, 0);
        long[] sorted = Arrays.copyOf(nanos, frames);
        Arrays.sort(sorted);
        long sum = 0;
        for (long n : sorted) sum += n;
        int p95 = Math.max(0, (int) Math.ceil(frames * 0.95) - 1);
        return new Report(seconds, frames, sum / 1e6 / frames, sorted[p95] / 1e6, sorted[frames - 1] / 1e6);
    }
}
```

**Files:**

**Interfaces:**

- Produces: a fragment shader with

    - `Plot`: x plot size along x, y plot size along z, z time in seconds, w night (0 day, 1 night)

```glsl
#version 330

#moj_import <minecraft:fog.glsl>

// The Veil fog: falling glowing smoke along the realm's four edges, drawn over the finished frame. Ported from the
// approved study (Veil Mist Studies, "Veil Fog, game version"). Per pixel and per wall: up to Shape.z steps through the
// smoke within 10 blocks of the wall, the rest in closed form (keep in step with VeilFogMath). Writes
// vec4(light, transmittance); the pipeline blends ONE, SRC_ALPHA, so the result is scene * T + L.

uniform sampler2D DepthSampler;
uniform sampler2D NoiseSampler;

layout(std140) uniform VeilFog {
    mat4 InvViewProj;   // screen to camera-relative position
    vec4 Camera;        // x, z relative to the plot's min corner; y in world blocks
    vec4 Plot;          // x, y: plot size along x and z; z: time in seconds; w: night 0..1
    vec4 Shape;         // x, y: top fade start and end y; z: steps
};

in vec2 texCoord;
out vec4 fragColor;

const float H = 2.4;          // the smoke halves about every 1.7 blocks out from its face
const float SIGMA = 0.8;      // density at the face
const float XMAX = 24.0;      // nothing worth drawing further out
const float NEAR = 10.0;      // stepped within this distance of the wall, closed form beyond
const float NEAR_PATH = 48.0; // at most this much of a view is stepped
const float SKY = 450.0;      // how far a sky pixel counts as
const float TILE = 34.0, ATLAS_W = 1088.0, SLICES = 32.0;

// ---- the noise atlas: 32 slices side by side, each pixel holding its slice and the next ----
vec4 atlas(vec3 c, out float f) {
    float s = fract(c.z) * SLICES;
    float k = min(floor(s), SLICES - 1.0);
    f = s - k;
    float u = (k * TILE + 1.5 + fract(c.x) * SLICES) / ATLAS_W;
    return textureLod(NoiseSampler, vec2(u, c.y + 0.5 / SLICES), 0.0);
}
float fieldA(vec3 c) { float f; vec4 n = atlas(c, f); return mix(n.r, n.g, f); }
float fieldB(vec3 c) { float f; vec4 n = atlas(c, f); return mix(n.b, n.a, f); }

// ---- the falling smoke at a point: along the wall, height, distance from the wall ----
// Two fields falling at 3.2 and 2 blocks a second, multiplied, so it changes as it falls; a sideways curl; tile sizes
// that never line up and slow sine bends travelling with each field, so the small atlas does not show as a pattern.
float smoke(float along, float y, float d, float t) {
    float qz = along + 1.2 * sin(y / 7.0 + t * 0.9 + d * 0.6);
    float ya = y + t * 3.2, yb = y + t * 2.0;
    vec2 wa = vec2(sin(qz / 41.0 + ya / 67.0), sin(qz / 29.3 - ya / 53.0 + 1.7));
    vec2 wb = vec2(sin(qz / 37.0 - yb / 59.0 + 4.1), sin(qz / 23.7 + yb / 71.0 + 0.6));
    float a = 0.25 + 0.5 * fieldA(vec3((qz + 5.0 * wa.x) / 19.4, (ya + 4.0 * wa.y) / 28.0, d / 12.0 + t * 0.0625));
    float b = 0.25 + 0.5 * fieldB(vec3((qz + 4.0 * wb.x) / 12.0 + 0.6, (yb + 3.0 * wb.y) / 19.1, d / 10.0 - t * 0.075));
    float broad = 0.07 * sin(qz / 97.0 + 0.4) + 0.05 * sin(qz / 61.0 - ya / 140.0);
    return clamp((a * b * 2.2 - 0.5) * 1.6 + 0.52 + broad, 0.0, 1.0);
}

// ---- closed forms (VeilFogMath) ----
float amount(float dc, float rx, float ta, float tb) {
    if (tb <= ta) return 0.0;
    float da = dc + rx * ta, db = dc + rx * tb;
    if (abs(rx) < 1e-3) return exp(-da / H) * (tb - ta);
    return (-H / rx) * (exp(-db / H) - exp(-da / H));
}
float middle(float dc, float rx, float ta, float tb) {
    if (abs(rx) < 1e-3 || tb <= ta) return 0.5 * (ta + tb);
    float da = dc + rx * ta, db = dc + rx * tb;
    float dm = -H * log(0.5 * (exp(-da / H) + exp(-db / H)));
    return clamp((dm - dc) / rx, ta, tb);
}
vec2 clipD(float lo, float hi, float dc, float rx, float ta, float tb) {
    if (abs(rx) < 1e-4) return (dc >= lo && dc <= hi) ? vec2(ta, tb) : vec2(0.0, -1.0);
    float a = (lo - dc) / rx, b = (hi - dc) / rx;
    return vec2(max(ta, min(a, b)), min(tb, max(a, b)));
}
float top(float y) { return 1.0 - smoothstep(Shape.x, Shape.y, y); }

// vanilla's distance fog over a point t blocks along the view
vec3 vanilla(vec3 c, float t, vec3 rd) {
    vec3 p = rd * t;
    float v = total_fog_value(t, max(length(p.xz), abs(p.y)), FogEnvironmentalStart, FogEnvironmentalEnd,
                              FogRenderDistanceStart, FogRenderDistanceEnd);
    return mix(c, FogColor.rgb, v * FogColor.a);
}

// ---- one wall: dc, rx as in VeilFogMath; ac, ar the same for the coordinate along the wall ----
// Returns vec4(light, transmittance); tin is where the view enters the smoke (1e9 if never).
vec4 wall(float dc, float rx, float ac, float ar, vec3 rd, float tEnd, float j, vec3 deep, vec3 lit, out float tin) {
    tin = 1e9;
    vec2 r = clipD(0.0, XMAX, dc, rx, 0.0, tEnd);
    if (r.y <= r.x) return vec4(0.0, 0.0, 0.0, 1.0);
    tin = r.x;
    float time = Plot.z;
    vec2 n = clipD(0.0, NEAR, dc, rx, r.x, r.y), f = clipD(NEAR, XMAX, dc, rx, r.x, r.y);
    float nEnd = min(n.y, n.x + NEAR_PATH);

    // the near stretch, stepped
    float Tn = 1.0; vec3 Ln = vec3(0.0);
    if (nEnd > n.x) {
        int count = int(Shape.z);
        float dt = (nEnd - n.x) / Shape.z;
        for (int i = 0; i < 16; i++) {
            if (i >= count) break;
            float t = n.x + (float(i) + j) * dt;
            float y = Camera.y + rd.y * t, d = dc + rx * t, al = ac + ar * t;
            float b = smoke(al, y, d, time);
            float out_ = d - 1.2 - 0.7 * sin(al / 8.0 + (y + time * 3.0) / 9.0) - (b - 0.4) * 5.0;  // thick smoke stands further out
            float sigma = SIGMA * exp(-max(out_, -3.0) / H) * smoothstep(0.25, 0.6, b) * top(y)
                        * mix(0.55, 1.0, smoothstep(0.0, 3.0, t));                                // thinner right around you
            float a = 1.0 - exp(-sigma * dt);
            vec3 c = mix(deep, lit, smoothstep(0.35, 0.8, b)) * mix(1.0, 0.75, smoothstep(0.0, 200.0, y));
            Ln += Tn * a * vanilla(c, t, rd);
            Tn *= 1.0 - a;
        }
    }
    // the rest of a long stretch along the wall, and the thin smoke further out: closed form
    float k = SIGMA * exp(1.2 / H) * 0.5;
    vec3 avg = mix(deep, lit, 0.3);
    float tr = middle(dc, rx, nEnd, n.y), tf = middle(dc, rx, f.x, f.y);
    float Tr = exp(-k * amount(dc, rx, nEnd, n.y) * top(Camera.y + rd.y * tr));
    float Tf = exp(-k * amount(dc, rx, f.x, f.y) * top(Camera.y + rd.y * tf));
    vec3 Lr = vanilla(avg, tr, rd) * (1.0 - Tr);
    vec3 Lf = vanilla(avg, tf, rd) * (1.0 - Tf);
    vec3 L = Ln + Tn * Lr; float T = Tn * Tr;                  // near, then its rest
    if (f.x < n.x) { L = Lf + Tf * L; } else { L = L + T * Lf; } // the outer smoke in front or behind
    T *= Tf;
    return vec4(L, T);
}

void main() {
    float z = texture(DepthSampler, texCoord).r;
    vec4 v = InvViewProj * vec4(texCoord * 2.0 - 1.0, z * 2.0 - 1.0, 1.0);
    vec3 rel = v.xyz / v.w;
    float tEnd = z >= 1.0 ? SKY : length(rel);
    vec3 rd = normalize(rel);

    float night = Plot.w;
    vec3 deep = mix(vec3(0.62, 0.62, 0.84), vec3(0.20, 0.17, 0.42), night);
    vec3 lit  = mix(vec3(0.96, 0.97, 1.00), vec3(0.62, 0.66, 1.00), night);
    float j = fract(52.9829189 * fract(dot(gl_FragCoord.xy, vec2(0.06711056, 0.00583715))));   // per-pixel jitter

    // the four walls: west (x = 0), east (x = size), north (z = 0), south (z = size)
    vec4 w[4]; float tin[4];
    w[0] = wall(Camera.x,          rd.x, Camera.z, rd.z, rd, tEnd, j, deep, lit, tin[0]);
    w[1] = wall(Plot.x - Camera.x, -rd.x, Camera.z, rd.z, rd, tEnd, j, deep, lit, tin[1]);
    w[2] = wall(Camera.z,          rd.z, Camera.x, rd.x, rd, tEnd, j, deep, lit, tin[2]);
    w[3] = wall(Plot.y - Camera.z, -rd.z, Camera.x, rd.x, rd, tEnd, j, deep, lit, tin[3]);

    // put them together front to back (at most two are ever non-empty: a corner)
    vec3 L = vec3(0.0); float T = 1.0;
    bool used[4] = bool[](false, false, false, false);
    for (int pass = 0; pass < 4; pass++) {
        int best = -1; float bt = 1e9;
        for (int i = 0; i < 4; i++) if (!used[i] && tin[i] < bt) { bt = tin[i]; best = i; }
        if (best < 0) break;
        used[best] = true;
        L += T * w[best].rgb;
        T *= w[best].a;
    }
    fragColor = vec4(L, T);
}
```

```java
package com.pocketkingdoms.client.veil;

/**
 * The per-frame numbers the Veil fog shader gets, worked out in doubles and handed over as small floats: positions
 * relative to the plot's min corner (exact even for a realm millions of blocks out) and time wrapped every hour. Plain
 * logic, unit tested; VeilFogRenderer writes them into the uniform block.
 */
public final class VeilFogUniforms {

    private static final long HOUR_TICKS = 72_000;

    private VeilFogUniforms() {}

    /** The camera's x and z relative to the plot's min corner, its y as is. */
    public static float[] camera(double camX, double camY, double camZ, int minX, int minZ) {
        return new float[] {(float) (camX - minX), (float) camY, (float) (camZ - minZ)};
    }

    /** Seconds of game time, wrapped every hour of ticks (one unseen jump an hour keeps float precision). */
    public static float time(long gameTime, float partialTick) {
        return (Math.floorMod(gameTime, HOUR_TICKS) + partialTick) / 20f;
    }

    /** 0 by day, 1 at night, from the level's sky darkening (15 minus the sky light). */
    public static float night(int skyDarken) {
        return Math.max(0f, Math.min(1f, skyDarken / 11f));
    }
}
```

```java
/**
 * The Veil fog: falling glowing smoke along the realm's four edges, one full-screen pass drawn at the very end of the
 * world render (LevelRendererMixin, RETURN of renderLevel). That is after any shader mod's final pass (Iris and its
 * Oculus ports overwrite the screen inside renderLevel), so the fog is never wiped; the hand is drawn later by
 * GameRenderer and never fogged. Reads the main depth texture (also Iris's scene depth) and the noise atlas, and blends
 * light and transmittance onto the frame. /pd test veilTime times it on the GPU (VeilFogTimer).
 */
public final class VeilFogRenderer {

    private static final Logger LOG = LogManager.getLogger();
    private static final int UBO_SIZE = 16 * 4 + 3 * 16;   // mat4 + 3 vec4, std140

    private static final RenderPipeline PIPELINE = RenderPipeline.builder(RenderPipelines.POST_PROCESSING_SNIPPET)
            .withLocation(Identifier.fromNamespaceAndPath(PocketKingdomsMod.MODID, "pipeline/veil_fog"))
            .withVertexShader("core/screenquad")
            .withFragmentShader(Identifier.fromNamespaceAndPath(PocketKingdomsMod.MODID, "core/veil_fog"))
            .withSampler("DepthSampler")
            .withSampler("NoiseSampler")
            .withUniform("VeilFog", UniformType.UNIFORM_BUFFER)
            .withUniform("Fog", UniformType.UNIFORM_BUFFER)
            .withBlend(new BlendFunction(SourceFactor.ONE, DestFactor.SRC_ALPHA, SourceFactor.ZERO, DestFactor.ONE))
            .build();

    private static @Nullable GpuTexture noise;
    private static @Nullable GpuTextureView noiseView;
    private static @Nullable MappableRingBuffer ubo;
    private static boolean ready, failed;
    private static final VeilFogTimer timer = new VeilFogTimer();
    private static final ArrayDeque<GpuQuery> pending = new ArrayDeque<>();

    private VeilFogRenderer() {}

    /** Reports /pd test veilTime in chat once its window closes. */
    public static void register() {
        TickEvent.ClientTickEvent.Post.BUS.addListener(e -> {
            VeilFogTimer.Report report = timer.poll(System.nanoTime());
            Minecraft mc = Minecraft.getInstance();
            if (report != null && mc.player != null) mc.player.displayClientMessage(Component.literal(report.describe()), false);
        });
    }

    /** /pd test veilTime: measure the next `seconds` of frames. */
    public static void measure(float seconds) {
        timer.start(seconds, System.nanoTime());
        Minecraft mc = Minecraft.getInstance();
        if (mc.player != null)
            mc.player.displayClientMessage(Component.literal(String.format(Locale.ROOT,
                    "Measuring the Veil fog for %.1f s. Look at the Veil as you normally would.", seconds)), false);
    }

    /** Called at the end of LevelRenderer.renderLevel with its camera, matrices and fog. */
    public static void render(Camera camera, Matrix4f viewRotation, Matrix4f projection, GpuBufferSlice fog, float partialTick) {
        collectTimings();
        Minecraft mc = Minecraft.getInstance();
        if (failed || mc.level == null || !mc.level.dimension().equals(PocketKingdomsMod.REALM_DIM)) return;
        if (!PocketKingdomsClientConfig.VEIL_FOG.get()) return;
        WardView.State ward = WardView.current();
        if (ward == null || WardSigilRenderer.skyHidden(camera)) return;
        try {
            if (!ready) setUp();
            if (failed) return;
            draw(mc, camera, ward.sync(), viewRotation, projection, fog, partialTick);
        } catch (RuntimeException e) {
            failed = true;
            LOG.warn("The Veil fog could not be drawn and is off until the game restarts; the Veil blocks still show.", e);
        }
    }

    private static void setUp() {
        ready = true;
        GpuDevice device = RenderSystem.getDevice();
        if (!device.precompilePipeline(PIPELINE).isValid()) {
            failed = true;
            LOG.warn("The Veil fog shader did not compile on this graphics card; the fog is off, the Veil blocks still show.");
            return;
        }
        int[] pixels = VeilNoise.atlas();
        try (NativeImage image = new NativeImage(VeilNoise.ATLAS_W, VeilNoise.ATLAS_H, false)) {
            for (int y = 0; y < VeilNoise.ATLAS_H; y++)
                for (int x = 0; x < VeilNoise.ATLAS_W; x++) image.setPixel(x, y, pixels[y * VeilNoise.ATLAS_W + x]);
            noise = device.createTexture("Veil fog noise", GpuTexture.USAGE_COPY_DST | GpuTexture.USAGE_TEXTURE_BINDING,
                    TextureFormat.RGBA8, VeilNoise.ATLAS_W, VeilNoise.ATLAS_H, 1, 1);
            device.createCommandEncoder().writeToTexture(noise, image);
        }
        noiseView = device.createTextureView(noise);
        ubo = new MappableRingBuffer(() -> "Veil fog", GpuBuffer.USAGE_UNIFORM | GpuBuffer.USAGE_MAP_WRITE, UBO_SIZE);
    }

    private static void draw(Minecraft mc, Camera camera, WardSyncS2C plot, Matrix4f viewRotation, Matrix4f projection,
                             GpuBufferSlice fog, float partialTick) {
        RenderTarget main = mc.getMainRenderTarget();
        if (main.getDepthTextureView() == null) return;
        Vec3 pos = camera.position();
        float[] cam = VeilFogUniforms.camera(pos.x, pos.y, pos.z, plot.minX(), plot.minZ());
        Matrix4f invViewProj = new Matrix4f(projection).mul(viewRotation).invert();
        int minY = mc.level.getMinY(), height = mc.level.getHeight();

        CommandEncoder encoder = RenderSystem.getDevice().createCommandEncoder();
        try (GpuBuffer.MappedView view = encoder.mapBuffer(ubo.currentBuffer(), false, true)) {
            Std140Builder.intoBuffer(view.data())
                    .putMat4f(invViewProj)
                    .putVec4(cam[0], cam[1], cam[2], 0)
                    .putVec4(plot.maxX() - plot.minX(), plot.maxZ() - plot.minZ(),
                            VeilFogUniforms.time(mc.level.getGameTime(), partialTick), VeilFogUniforms.night(mc.level.getSkyDarken()))
                    .putVec4((float) VeilFogMath.fadeStart(minY, height), (float) VeilFogMath.fadeEnd(minY, height),
                            PocketKingdomsClientConfig.VEIL_FOG_STEPS.get(), 0);
        }
        GpuQuery query = timer.measuring() && pending.size() < 8 ? encoder.timerQueryBegin() : null;
        try (RenderPass pass = encoder.createRenderPass(() -> "Veil fog", main.getColorTextureView(), OptionalInt.empty(),
                null, OptionalDouble.empty())) {
            pass.setPipeline(PIPELINE);
            pass.setUniform("VeilFog", ubo.currentBuffer());
            pass.setUniform("Fog", fog);
            pass.bindTexture("DepthSampler", main.getDepthTextureView(), RenderSystem.getSamplerCache().getClampToEdge(FilterMode.NEAREST));
            pass.bindTexture("NoiseSampler", noiseView, RenderSystem.getSamplerCache().getRepeat(FilterMode.LINEAR));
            pass.draw(0, 3);
        }
        if (query != null) {
            encoder.timerQueryEnd(query);
            pending.add(query);
        }
        ubo.rotate();
    }

    /** GPU timings arrive a frame or two late: hand over the ones that are ready. */
    private static void collectTimings() {
        while (!pending.isEmpty()) {
            OptionalLong value = pending.peek().getValue();
            if (value.isEmpty()) return;
            timer.frame(value.getAsLong());
            pending.poll().close();
        }
    }
}
```
